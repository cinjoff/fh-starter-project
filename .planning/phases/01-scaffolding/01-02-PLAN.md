## Task 2: ActionState type, SubmitButton, and dates utility

### 2a: ActionState<T> type

Create `src/lib/actions/types.ts`:

```typescript
export type ActionState<T = void> = {
  success: boolean
  message?: string
  errors?: Record<string, string[]>
  data?: T
}
```

This is the standard return type for all server actions. Forms use `errors` for field-level validation, `message` for general feedback.

### 2b: SubmitButton component

Create `src/components/submit-button.tsx`:

```tsx
"use client"

import { useFormStatus } from "react-dom"
import { Button, type buttonVariants } from "@/components/ui/button"
import type { VariantProps } from "class-variance-authority"

type SubmitButtonProps = React.ComponentProps<"button"> &
  VariantProps<typeof buttonVariants> & {
    pendingText?: string
  }

export function SubmitButton({
  children,
  pendingText = "Submitting...",
  ...props
}: SubmitButtonProps) {
  const { pending } = useFormStatus()

  return (
    <Button type="submit" disabled={pending} {...props}>
      {pending ? pendingText : children}
    </Button>
  )
}
```

### 2c: Date utility

Create `src/lib/dates.ts`:

```typescript
import { format, formatDistanceToNow, parseISO } from "date-fns"

export function formatDate(date: string | Date): string {
  const d = typeof date === "string" ? parseISO(date) : date
  return format(d, "MMM d, yyyy")
}

export function formatRelative(date: string | Date): string {
  const d = typeof date === "string" ? parseISO(date) : date
  return formatDistanceToNow(d, { addSuffix: true })
}
```
